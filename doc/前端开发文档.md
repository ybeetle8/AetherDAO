# AetherDAO 前端开发文档

## 一、项目概述

### 1.1 产品简介

AetherDAO 是一个运行在 BSC（币安智能链）上的 DeFi 平台，围绕 **AE 代币** 构建，包含以下核心功能：

- **AE 代币交易** — 通过 PancakeSwap 买卖 AE，自动收取税费用于生态建设
- **USDT 质押理财** — 用户质押 USDT 获得每日复利收益
- **推荐人体系** — 多级推荐关系与团队等级奖励系统
- **LP 流动性挖矿** — 质押 LP Token 赚取 USDT 奖励

## 三、AE 代币模块

### 3.1 功能概述

AE 是平台原生 ERC20 代币，通过 PancakeSwap 与 USDT 组成交易对。买卖时自动收取税费。

**买入税 3%**：2% 节点分红 + 1% 社区分红
**卖出固定税 3%**：1.5% 营销基金 + 1.5% 销毁
**卖出利润税**：盈利部分的 25%

### 3.3 AE 买卖交易

AE 的买卖通过 **PancakeSwap Router** 进行，不需要直接调用 AE 合约。AE 合约的 `_transfer` 函数会自动检测买卖行为并扣税。

#### 买入 AE（USDT → AE）

```javascript
async function buyAE(signer, usdtAmount) {
  const router = new ethers.Contract(ADDRESSES.PANCAKE_ROUTER, ROUTER_ABI, signer);
  const amountIn = parseAmount(usdtAmount);
  const userAddress = await signer.getAddress();

  // 1. 授权 USDT 给 Router
  await approveUSDT(signer, ADDRESSES.PANCAKE_ROUTER, amountIn);

  // 2. 构建交易路径
  const path = [ADDRESSES.USDT, ADDRESSES.AE];

  // 3. 获取预期输出量
  const amounts = await router.getAmountsOut(amountIn, path);
  const expectedOut = amounts[1];

  // 4. 设置滑点容忍度（建议 5-10%，因为有 3% 买入税）
  const slippage = 10; // 10%
  const minOut = expectedOut * BigInt(100 - slippage) / 100n;

  // 5. 执行买入
  const deadline = Math.floor(Date.now() / 1000) + 300; // 5分钟超时
  const tx = await router.swapExactTokensForTokensSupportingFeeOnTransferTokens(
    amountIn,
    minOut,
    path,
    userAddress,
    deadline
  );
  return await tx.wait();
}
```

> **注意**: 必须使用 `swapExactTokensForTokensSupportingFeeOnTransferTokens` 方法，因为 AE 代币有转账税。使用普通的 `swapExactTokensForTokens` 会失败。

#### 卖出 AE（AE → USDT）

```javascript
async function sellAE(signer, aeAmount) {
  const router = new ethers.Contract(ADDRESSES.PANCAKE_ROUTER, ROUTER_ABI, signer);
  const ae = new ethers.Contract(ADDRESSES.AE, AE_ABI, signer);
  const amountIn = parseAmount(aeAmount);
  const userAddress = await signer.getAddress();

  // 1. 检查冷却期（买入后 10 秒内不能卖出）
  const [, lastBuy] = await ae.getUserInvestment(userAddress);
  const now = Math.floor(Date.now() / 1000);
  if (now - Number(lastBuy) < 10) {
    throw new Error('冷却期内，请等待 10 秒后再卖出');
  }

  // 2. 授权 AE 给 Router
  const tx1 = await ae.approve(ADDRESSES.PANCAKE_ROUTER, amountIn);
  await tx1.wait();

  // 3. 构建交易路径
  const path = [ADDRESSES.AE, ADDRESSES.USDT];

  // 4. 获取预期输出（注意：实际到手会因税费减少）
  const amounts = await router.getAmountsOut(amountIn, path);
  const expectedOut = amounts[1];

  // 5. 设置较高滑点（固定 3% 税 + 可能的 25% 利润税）
  const slippage = 30; // 30%（保守值，实际根据盈亏情况调整）
  const minOut = expectedOut * BigInt(100 - slippage) / 100n;

  // 6. 执行卖出
  const deadline = Math.floor(Date.now() / 1000) + 300;
  const tx = await router.swapExactTokensForTokensSupportingFeeOnTransferTokens(
    amountIn,
    minOut,
    path,
    userAddress,
    deadline
  );
  return await tx.wait();
}
```

#### 卖出利润预估

```javascript
// 预估卖出后的利润和税费
async function estimateSellProfit(userAddress, aeAmount) {
  const ae = new ethers.Contract(ADDRESSES.AE, AE_ABI, provider);
  const router = new ethers.Contract(ADDRESSES.PANCAKE_ROUTER, ROUTER_ABI, provider);

  // 获取用户历史投资
  const [investment] = await ae.getUserInvestment(userAddress);

  // 预估可获得的 USDT
  const path = [ADDRESSES.AE, ADDRESSES.USDT];
  const amounts = await router.getAmountsOut(parseAmount(aeAmount), path);
  const estimatedUSDT = amounts[1];

  // 计算固定税（3%）
  const fixedTax = estimatedUSDT * 3n / 100n;

  // 计算利润
  const profit = estimatedUSDT > investment
    ? estimatedUSDT - investment
    : 0n;

  // 利润税（25%）
  const profitTax = profit * 25n / 100n;

  return {
    estimatedUSDT: formatAmount(estimatedUSDT),
    fixedTax: formatAmount(fixedTax),
    profit: formatAmount(profit),
    profitTax: formatAmount(profitTax),
    netReceived: formatAmount(estimatedUSDT - fixedTax - profitTax),
  };
}
```

## 四、USDT 质押模块

### 4.1 功能概述

用户将 USDT 质押到 Staking 合约，获得每日复利收益。质押后系统自动将 USDT 添加流动性。

**质押规则**：
- 最低质押：100 USDT
- 单次上限：1,000 USDT
- 总质押上限：每用户 10,000 USDT
- 7 天期质押每个用户仅可使用一次

**质押期限与日收益率**：

| 索引 (stakeIndex) | 期限 | 日收益率 | 说明 |
|:---:|:---:|:---:|:---:|
| 0 | 7 天 | 0.6% | 每用户限一次 |
| 1 | 30 天 | 0.9% | - |
| 2 | 90 天 | 1.1% | - |
| 3 | 180 天 | 1.5% | - |
| 4 | 365 天 | 2.0% | - |

### 4.2 前置条件 — 绑定推荐人

**质押前必须先绑定推荐人**，否则 `stake()` 调用会失败。

```javascript
// 检查是否已绑定推荐人
const isBound = await staking.isBindReferral(userAddress);

// 绑定推荐人（一次性操作，不可更改）
async function bindReferral(signer, referrerAddress) {
  const staking = new ethers.Contract(ADDRESSES.STAKING, STAKING_ABI, signer);
  const tx = await staking.lockReferral(referrerAddress);
  return await tx.wait();
}

// 获取用户的推荐人
const referrer = await staking.getReferral(userAddress);
```

**绑定规则**：
- 推荐人必须已经绑定过推荐关系（或为 Root 节点）
- 不能自我推荐
- 不允许循环引用
- 一旦绑定不可更改

```javascript
async function stakeUSDT(signer, amount, periodIndex) {
  const staking = new ethers.Contract(ADDRESSES.STAKING, STAKING_ABI, signer);
  const amountWei = parseAmount(amount);
  const userAddress = await signer.getAddress();

  // 5. 执行质押
  // 注意: amount 参数类型为 uint160
  const tx = await staking.stake(amountWei, periodIndex);
  return await tx.wait();
}
```

### 4.5 复利收益计算公式

前端可以用以下公式在本地计算预期收益，用于页面展示：

```javascript
/**
 * 计算复利收益
 * @param {number} principal 本金 (USDT)
 * @param {number} dailyRate 日利率（如 0.9% 传入 0.009）
 * @param {number} days 天数
 * @returns {object} { totalValue, interest }
 */
function calculateCompoundInterest(principal, dailyRate, days) {
  const totalValue = principal * Math.pow(1 + dailyRate, days);
  const interest = totalValue - principal;
  return {
    totalValue: totalValue.toFixed(2),
    interest: interest.toFixed(2),
  };
}

// 各期限的日利率常量
const DAILY_RATES = {
  0: 0.006,   // 7天  → 0.6%
  1: 0.009,   // 30天 → 0.9%
  2: 0.011,   // 90天 → 1.1%
  3: 0.015,   // 180天→ 1.5%
  4: 0.020,   // 365天→ 2.0%
};

// 示例：1000 USDT 质押 30 天
const result = calculateCompoundInterest(1000, DAILY_RATES[1], 30);
// result.totalValue ≈ "1308.65"
// result.interest ≈ "308.65"
```

```javascript
    nextTierThreshold, progressToNextTier] =

    parseAmount(300),    // V2
    parseAmount(600),    // V3
    parseAmount(1000),   // V4

    parseAmount(6000),   // V6
```

#### 差额分配机制说明

团队奖励采用差额制，前端可用于展示预估奖励分配：

```javascript
/**
 * 模拟差额分配（仅用于 UI 展示）
 * @param {number} interestAmount 利息金额 (USDT)
 * @param {Array} referralChainTiers 推荐链上各节点的等级 [{address, tier}]
 */
function simulateDifferentialReward(interestAmount, referralChainTiers) {
  const teamRewardPool = interestAmount * 0.35; // 最高 35%
  const tierRates = [0, 3, 7, 11, 15, 19, 23, 27, 31, 35]; // V0~V9

  let cumulativeRate = 0;
  const distribution = [];

  for (const node of referralChainTiers) {
    const nodeRate = tierRates[node.tier] || 0;
    if (nodeRate > cumulativeRate) {
      const diff = nodeRate - cumulativeRate;
      const reward = teamRewardPool * diff / 35;
      distribution.push({
        address: node.address,
        tier: node.tier,
        rate: diff,
        reward: reward.toFixed(2),
      });
      cumulativeRate = nodeRate;
    }
  }

  // 剩余归根节点
  if (cumulativeRate < 35) {
    const remaining = teamRewardPool * (35 - cumulativeRate) / 35;
    distribution.push({
      address: 'Root',
      tier: 'Root',
      rate: 35 - cumulativeRate,
      reward: remaining.toFixed(2),
    });
  }

  return distribution;
}
```

```javascript
async function getStakeList(userAddress) {
  const staking = new ethers.Contract(ADDRESSES.STAKING, STAKING_ABI, provider);
  const count = await staking.stakeCount(userAddress);
  const periods = [7, 30, 90, 180, 365]; // 天
  const dailyRates = [0.006, 0.009, 0.011, 0.015, 0.020];

  const stakes = [];
  for (let i = 0; i < count; i++) {
    const record = await staking.userStakeRecord(userAddress, i);
    if (!record.status) continue; // 跳过已赎回

    const stakeTime = Number(record.stakeTime);
    const amount = record.amount;
    const periodIndex = record.stakeIndex;
    const periodDays = periods[periodIndex];
    const endTime = stakeTime + periodDays * 86400;
    const now = Math.floor(Date.now() / 1000);

    // 已质押天数
    const elapsedDays = Math.min(
      Math.floor((now - stakeTime) / 86400),
      periodDays
    );

    // 当前收益
    const reward = await staking.rewardOfSlot(userAddress, i);

    // 可提取利息
    const availableInterest = await staking.getAvailableInterest(userAddress, i);
    const withdrawnInterest = await staking.getWithdrawnInterest(userAddress, i);

    // 是否可赎回
    const canWithdraw = now >= endTime;

    stakes.push({
      index: i,
      amount: formatAmount(amount),
      periodDays,
      dailyRate: (dailyRates[periodIndex] * 100).toFixed(1) + '%',
      stakeTime: new Date(stakeTime * 1000).toLocaleString(),
      endTime: new Date(endTime * 1000).toLocaleString(),
      elapsedDays,
      totalReward: formatAmount(reward),
      availableInterest: formatAmount(availableInterest),
      withdrawnInterest: formatAmount(withdrawnInterest),
      canWithdraw,
      timeRemaining: canWithdraw ? 0 : endTime - now,
    });
  }
  return stakes;
}
```

```javascript
function parseContractError(error) {
  // ethers.js v6 自定义错误解析
  if (error.data) {
    try {
      // 尝试匹配已知错误
      for (const [errorName, message] of Object.entries(ERROR_MESSAGES)) {
        if (error.message?.includes(errorName) || error.data?.includes(errorName)) {
          return message;
        }
      }
    } catch (e) {}
  }

  // MetaMask 用户拒绝
  if (error.code === 4001 || error.code === 'ACTION_REJECTED') {
    return '用户取消了交易';
  }

  // Gas 不足
  if (error.code === -32603 || error.message?.includes('insufficient funds')) {
    return 'BNB 余额不足以支付 Gas 费';
  }

  return error.reason || error.message || '交易失败，请重试';
```

### 8.2 交易确认流程

```javascript
// 统一的交易发送与等待函数
async function sendTransaction(txPromise, description) {
  try {
    // 1. 发送交易
    const tx = await txPromise;
    console.log(`${description} 交易已发送: ${tx.hash}`);

    // 2. 等待确认（BSC 通常 3 秒出块）
    const receipt = await tx.wait();

    if (receipt.status === 1) {
      console.log(`${description} 成功`);
      return { success: true, receipt };
    } else {
      return { success: false, error: '交易被回滚' };
    }
  } catch (error) {
    return { success: false, error: parseContractError(error) };
  }
}
```
