```solidity
function stake(uint160 _amount, uint8 _stakeIndex) external onlyEOA {
    _validateStakeParameters(_amount, _stakeIndex);

    // Check 7-day stake usage limit
    if (_stakeIndex == 0) {
        require(!hasUsed7DayStake[msg.sender], "7-day stake can only be used once");
        hasUsed7DayStake[msg.sender] = true;
        emit FirstTime7DayStakeUsed(msg.sender, block.timestamp);
    }

    _swapAndAddLiquidity(_amount);
    address user = msg.sender;
    _mintStakeRecord(user, _amount, _stakeIndex);
}
```

#### 新增管理函数
```solidity
/**
 * @notice Reset a user's 7-day stake usage status (owner only)
 * @param user User address to reset
 */
function reset7DayStakeUsage(address user) external onlyOwner {
    require(user != address(0), "Invalid user address");
    hasUsed7DayStake[user] = false;
    emit Stake7DayUsageReset(user, block.timestamp);
}

/**
 * @notice Batch reset multiple users' 7-day stake usage status (owner only)
 * @param users Array of user addresses to reset
 */
function batchReset7DayStakeUsage(address[] calldata users) external onlyOwner {
    for (uint256 i = 0; i < users.length; i++) {
        if (users[i] != address(0)) {
            hasUsed7DayStake[users[i]] = false;
            emit Stake7DayUsageReset(users[i], block.timestamp);
        }
    }
}
```
**位置**: 第1429-1450行

## 🎯 功能说明

### 核心功能
1. **限制检查**: 用户尝试使用7天质押（stakeIndex = 0）时，系统会检查 `hasUsed7DayStake[user]`
2. **首次使用**: 如果用户未使用过，标记为已使用并触发 `FirstTime7DayStakeUsed` 事件
3. **重复使用**: 如果用户已使用过，交易会回滚并显示错误信息："7-day stake can only be used once"

### 查询功能
- `has7DayStakeBeenUsed(address user)`: 查询指定用户是否已使用7天质押
- 前端可以调用此函数来禁用按钮或显示提示

### 管理功能（仅 Owner）
- `reset7DayStakeUsage(address user)`: 重置单个用户的使用状态
- `batchReset7DayStakeUsage(address[] users)`: 批量重置多个用户的使用状态

## 🔒 安全特性

### 1. 访问控制
- ✅ 重置功能仅限 Owner 调用（`onlyOwner` 修饰符）
- ✅ 查询功能为 `view` 函数，不消耗 gas

### 2. 输入验证
- ✅ 批量重置时跳过零地址
- ✅ 单个重置时检查地址有效性

### 3. 事件记录
- ✅ 首次使用时触发 `FirstTime7DayStakeUsed` 事件
- ✅ 重置时触发 `Stake7DayUsageReset` 事件
- ✅ 所有操作可追溯

### 4. 遵循 Checks-Effects-Interactions 模式
```solidity

// 2. Effects

// 3. Interactions
_swapAndAddLiquidity(_amount);
```
